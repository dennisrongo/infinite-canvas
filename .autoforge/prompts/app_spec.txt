<project_specification>
  <project_name>Infinite Canvas</project_name>

  <overview>
    An infinite canvas note-taking application where users can create, organize, and connect markdown-enabled notes on an unlimited pan/zoom canvas. Users can register for accounts, create multiple canvas projects organized in folders, link notes visually with drag connectors, and enjoy a beautiful, responsive experience with light/dark themes.
  </overview>

  <technology_stack>
    <frontend>
      <framework>Next.js 16</framework>
      <styling>Tailwind CSS</styling>
      <canvas_library>React Flow (xyflow)</canvas_library>
      <editor>Markdown editor with rich text toolbar</editor>
      <state_management>React Context + Zustand for canvas state</state_management>
    </frontend>
    <backend>
      <runtime>Next.js 16 API Routes / Server Actions</runtime>
      <database>NeonDB (PostgreSQL)</database>
      <orm>Prisma ORM</orm>
      <authentication>NextAuth.js or custom JWT implementation</authentication>
    </backend>
    <communication>
      <api>RESTful API with Next.js API Routes / Server Actions</api>
      <realtime>Not required for MVP (no collaboration)</realtime>
    </communication>
  </technology_stack>

  <database_schema>
    <tables>
      <User>
        - id (UUID, primary key)
        - email (String, unique, indexed)
        - password_hash (String)
        - display_name (String, optional)
        - created_at (DateTime)
        - updated_at (DateTime)
        - last_login (DateTime, optional)
      </User>

      <Folder>
        - id (UUID, primary key)
        - user_id (UUID, foreign key to User)
        - name (String)
        - created_at (DateTime)
        - updated_at (DateTime)
        - Index on user_id
      </Folder>

      <Canvas>
        - id (UUID, primary key)
        - user_id (UUID, foreign key to User)
        - folder_id (UUID, foreign key to Folder, nullable - for root canvases)
        - name (String)
        - created_at (DateTime)
        - updated_at (DateTime)
        - Index on user_id
        - Index on folder_id
      </Canvas>

      <Note>
        - id (UUID, primary key)
        - canvas_id (UUID, foreign key to Canvas)
        - title (String)
        - content (Text, markdown content)
        - position_x (Float - x coordinate on canvas)
        - position_y (Float - y coordinate on canvas)
        - width (Float - node width)
        - height (Float - node height)
        - created_at (DateTime)
        - updated_at (DateTime)
        - Index on canvas_id
        - Index on title (for search)
      </Note>

      <NoteConnection>
        - id (UUID, primary key)
        - canvas_id (UUID, foreign key to Canvas)
        - source_note_id (UUID, foreign key to Note)
        - target_note_id (UUID, foreign key to Note)
        - created_at (DateTime)
        - Index on canvas_id
        - Index on source_note_id
        - Index on target_note_id
      </NoteConnection>

      <Image>
        - id (UUID, primary key)
        - note_id (UUID, foreign key to Note)
        - storage_path (String - URL or path to stored image)
        - file_name (String)
        - mime_type (String)
        - size_bytes (Integer)
        - created_at (DateTime)
        - Index on note_id
      </Image>

      <PasswordResetToken>
        - id (UUID, primary key)
        - user_id (UUID, foreign key to User)
        - token (String, unique, hashed)
        - expires_at (DateTime)
        - created_at (DateTime)
        - Index on token
      </PasswordResetToken>
    </tables>
  </database_schema>

  <api_endpoints_summary>
    <Authentication>
      - POST /api/auth/register - Create new user account
      - POST /api/auth/login - Authenticate user and create session
      - POST /api/auth/logout - End user session
      - POST /api/auth/reset-password-request - Request password reset email
      - POST /api/auth/reset-password - Reset password with token
      - GET /api/auth/me - Get current authenticated user
    </Authentication>

    <Canvas_Management>
      - GET /api/canvases - Get all canvases for current user
      - POST /api/canvases - Create new canvas
      - GET /api/canvases/:id - Get single canvas with notes and connections
      - PUT /api/canvases/:id - Update canvas name or folder
      - DELETE /api/canvases/:id - Delete canvas
    </Canvas_Management>

    <Folder_Management>
      - GET /api/folders - Get all folders for current user
      - POST /api/folders - Create new folder
      - PUT /api/folders/:id - Update folder name
      - DELETE /api/folders/:id - Delete folder
    </Folder_Management>

    <Note_Management>
      - GET /api/canvases/:canvasId/notes - Get all notes in a canvas
      - POST /api/canvases/:canvasId/notes - Create new note
      - GET /api/notes/:id - Get single note
      - PUT /api/notes/:id - Update note content or position
      - DELETE /api/notes/:id - Delete note
      - POST /api/notes/:id/duplicate - Create copy of note
    </Note_Management>

    <Connection_Management>
      - GET /api/canvases/:canvasId/connections - Get all connections in a canvas
      - POST /api/canvases/:canvasId/connections - Create connection between notes
      - DELETE /api/connections/:id - Delete connection
    </Connection_Management>

    <Image_Management>
      - POST /api/notes/:noteId/images - Upload image to note
      - DELETE /api/images/:id - Delete image
    </Image_Management>

    <Search>
      - GET /api/search?q=query - Search across all canvases and notes
      - GET /api/canvases/:canvasId/search?q=query - Search within specific canvas
    </Search>

    <User_Settings>
      - GET /api/user/settings - Get user preferences (theme, etc.)
      - PUT /api/user/settings - Update user preferences
      - DELETE /api/user/account - Delete user account
    </User_Settings>
  </api_endpoints_summary>
